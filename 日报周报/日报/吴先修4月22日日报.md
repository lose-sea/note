## 今日算法题

### 题目1: [142. 环形链表 II](https://leetcode.cn/problems/linked-list-cycle-ii/)

![image-20260422213546800](/Users/lose_sea/Desktop/pintures/image-20260422213546800.png)

```objc

//  // 哈希表
// class Solution {
// public:
//     ListNode *detectCycle(ListNode *head) {
//         unordered_set<ListNode*> visited; 
//         ListNode* fast = head; 
//         ListNode* slow = head; 
//         while (slow != nullptr) {
//             if (visited.count(slow)) {
//                 return slow; 
//             } 
//             visited.insert(slow); 
//             slow = slow->next; 
//         } 
//         return nullptr; 
//     }
// };

 // 双指针
class Solution {
public:
    ListNode *detectCycle(ListNode *head) {
        ListNode* fast = head; 
        ListNode* slow = head; 
        while (fast != nullptr && fast->next != nullptr) {
            fast = fast->next->next; 
            slow = slow->next;  
            if (fast == slow) {
                fast = head;  
                while (fast != slow) {
                    fast = fast->next; 
                    slow = slow->next;  
                } 
                return fast; 
            }
        } 
        return nullptr; 
    }
};
```
