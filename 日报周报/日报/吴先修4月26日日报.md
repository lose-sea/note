### 题目1:

![image-20260426223852069](C:\Users\Lenovo\AppData\Roaming\Typora\typora-user-images\image-20260426223852069.png)

```objc
#define ll int 
class Solution {
public:    
    ll result; 
    ll dir[2][4] = {
        {0, -1, 0, 1}, 
        {-1, 0, 1, 0}
    }; 
    void bfs(vector<vector<ll>>& grid, vector<vector<bool>>& visited, ll x, ll y) {
        if (grid[x][y] == 0 || visited[x][y] == true) {
            return; 
        } 
        queue<pair<ll, ll>> que; 
        que.push({x, y}); 
        visited[x][y] = true;   
        while (!que.empty()) { 
            x = que.front().first; 
            y = que.front().second; 
            que.pop(); 
            for (ll i = 0; i < 4; i++) {
                ll nx = x + dir[0][i]; 
                ll ny = y + dir[1][i]; 
                if (nx < 0 || nx >= grid.size() || ny < 0 || ny >= grid[0].size() || grid[nx][ny] == 0) {
                    result++;  
                    continue; 
                } 
                if (grid[nx][ny] == 1 && visited[nx][ny] == false) {
                    que.push({nx, ny});   
                    visited[nx][ny] = true; 
                }
            } 
        }
    }
    int islandPerimeter(vector<vector<int>>& grid) { 
        result = 0;  
        vector<vector<bool>> visited(grid.size(), vector<bool>(grid[0].size(), false)); 
        for (ll i = 0; i < grid.size(); i++) {
            for (ll j = 0; j < grid[0].size(); j++) {
                bfs(grid, visited, i, j); 
            }
        } 
        return result; 
    }
};
```
