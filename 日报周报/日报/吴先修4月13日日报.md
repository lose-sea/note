```objc
#define ll long long 
class Solution {
public:
    void rotate(vector<vector<int>>& matrix) { 
        ll n = matrix.size(); 
        for (ll i = 0; i < (n / 2); i++) {
            for (ll j = 0; j < (n + 1) / 2; j++) {
                ll temp = matrix[i][j]; 
                matrix[i][j] = matrix[n - j - 1][i]; 
                matrix[n - j - 1][i] = matrix[n - i - 1][n - j - 1]; 
                matrix[n - i - 1][n - j - 1] = matrix[j][n - i - 1]; 
                matrix[j][n - i - 1] = temp; 
            }
        } 
    }
};
```
