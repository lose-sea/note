int DIRS[128][2];

int init = [] {
    DIRS['L'][0] = DIRS['D'][1] = -1;
    DIRS['R'][0] = DIRS['U'][1] = 1;
    return 0;
}();

class Solution {
public:
    int distinctPoints(string s, int k) {
        int n = s.size();
        unordered_set<long long> st;
        st.insert(1LL * n << 32 | n); // 第一个窗口
        int x = 0, y = 0;
        for (int i = k; i < n; i++) {
            char in = s[i], out = s[i - k];
            x += DIRS[in][0] - DIRS[out][0];
            y += DIRS[in][1] - DIRS[out][1];
            st.insert(1LL * (x + n) << 32 | (y + n));
        }
        return st.size();
    }
};
