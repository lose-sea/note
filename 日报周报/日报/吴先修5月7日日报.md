## 今日算法题

### 题目1: [543. 二叉树的直径](https://leetcode.cn/problems/diameter-of-binary-tree/) 

![image-20260507215105671](/Users/lose_sea/Desktop/pintures/image-20260507215105671.png)

```objc
 #define ll long long
class Solution {
public: 
    ll result; 
    int def(TreeNode* root) {
        if (root == nullptr) {
            return 0; 
        } 
        ll l = def(root->left); 
        ll r = def(root->right); 
        result = max(result, l + r + 1); 
        return max(l, r) + 1; 
    }
    int diameterOfBinaryTree(TreeNode* root) {
        result = 1; 
        def(root); 
        return result - 1; 
    }
};
```
