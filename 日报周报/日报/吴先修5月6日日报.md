## 今日算法题

### 题目1: [104. 二叉树的最大深度](https://leetcode.cn/problems/maximum-depth-of-binary-tree/) 

![image-20260506215131725](/Users/lose_sea/Desktop/pintures/image-20260506215131725.png)

```objc
#define ll long long  
// // 深度优先搜索
// class Solution {
// public: 
//     int maxDepth(TreeNode* root) {
//         if (root == nullptr) { 
//             return 0; 
//         }  
//         ll left = maxDepth(root->left); 
//         ll right = maxDepth(root->right); 
//         return max(left, right) + 1; 
//     }
// }; 

// 广度优先搜索
class Solution {
public: 
    int maxDepth(TreeNode* root) { 
        ll result = 0; 
        if (root == nullptr) { 
            return result; 
        }  
        queue<TreeNode*> que; 
        que.push(root); 
        while (!que.empty()) {  
            ll num = que.size(); 
            for (ll i = 0; i < num; i++) {
                TreeNode* curr = que.front(); 
                que.pop(); 
                if (curr->left) {
                    que.push(curr->left); 
                } 
                if (curr->right) {
                    que.push(curr->right); 
                }  
            }
            result++; 
        } 
        return result; 
    }
}; 
```
