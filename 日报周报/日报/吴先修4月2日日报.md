## 今日算法题

### 题目1: [76. 最小覆盖子串](https://leetcode.cn/problems/minimum-window-substring/) 

![image-20260402214204090](/Users/lose_sea/Library/Application Support/typora-user-images/image-20260402214204090.png)

```objc
class Solution {
public: 
    string minWindow(string s, string t) { 
        if (t.size() > s.size()) {
            return ""; 
        }
        int diff[128] = {}; 
        int kinds = 0; 
        for (char c : t) { 
            if (diff[c] == 0) {
                kinds++; 
            }
            diff[c]--; 
        } 
        int m = s.size(); 
        int ans_left = -1; 
        int ans_right = s.size(); 
        int left = 0; 
        for (int right = 0; right < m; right++) {
            char c = s[right]; 
            diff[c]++; 
            if (diff[c] == 0) {
                kinds--; 
            } 
            while (kinds == 0) {
                if (right - left < ans_right - ans_left) {
                    ans_left = left; 
                    ans_right = right; 
                } 
                char x = s[left];  
                if (diff[x] == 0) {
                    kinds++; 
                } 
                diff[x]--; 
                left++; 
            }
        }
        return ans_left < 0 ? "" : s.substr(ans_left, ans_right - ans_left + 1); 
    }
};
```
