## 今日算法题

### 题目1:  [77. 组合](https://leetcode.cn/problems/combinations/) 

![image-20260508214547813](/Users/lose_sea/Desktop/pintures/image-20260508214547813.png)

```objc
class Solution {
public: 
    vector<vector<int>> result; 
    vector<int> path;  

    void backtracking(int n, int k, int startIndex) {
        if (path.size() == k) {
            result.push_back(path); 
            return; 
        } 
        for (int i = startIndex; i <= n - (k - path.size()) + 1; i++) {
            path.push_back(i); 
            backtracking(n, k, i + 1); 
            path.pop_back(); 
        } 
    }

    vector<vector<int>> combine(int n, int k) {  
        result.clear(); 
        path.clear(); 
        backtracking(n, k, 1); 
        return result; 
    }
};
```
