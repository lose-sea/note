### 题目1:[200. 岛屿数量](https://leetcode.cn/problems/number-of-islands/)

![image-20260423140416312](C:\Users\Lenovo\AppData\Roaming\Typora\typora-user-images\image-20260423140416312.png)

```objc
#define ll long long 
class Solution {
public:  
    ll dir[2][4] = {
        {-1, 0, 1, 0}, 
        {0, -1, 0, 1}
    }; 
    void dfs(vector<vector<char>>& grid, vector<vector<bool>>& visited, ll x, ll y) {
        if (visited[x][y] == true || grid[x][y] == '0') {
            return; 
        }  
        visited[x][y] = true; 
        for (ll i = 0; i < 4; i++) {
            ll nextx = x + dir[0][i]; 
            ll nexty = y + dir[1][i];  
            if (nextx < 0 || nextx >= grid.size() || nexty < 0 || nexty >= grid[0].size() || grid[nextx][nexty] == '0') {
                continue; 
            } else {
                dfs(grid, visited, nextx, nexty); 
            }
        }
    } 

    int numIslands(vector<vector<char>>& grid) {
        vector<vector<bool>> visited(grid.size(), vector<bool>(grid[0].size(), false));  
        ll result = 0; 
        for (ll i = 0; i < grid.size(); i++) {
            for (ll j = 0; j < grid[0].size(); j++) {
                if (grid[i][j] == '0' || visited[i][j] == true) {
                    continue; 
                } 
                dfs(grid, visited, i, j);  
                result++; 
            }
        } 
        return result; 
    }
};
```
