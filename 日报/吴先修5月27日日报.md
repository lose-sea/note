## 今日算法题

### 题目1:  [112. 路径总和](https://leetcode.cn/problems/path-sum/) 

![image-20260527214848805](/Users/lose_sea/Desktop/pintures/image-20260527214848805.png)

```objc

class Solution {
public: 
    bool result = false; 
    int curr = 0; 
    void traversal(TreeNode* root, int targetSum) {
        if (result == true) {
            return; 
        }  
        curr += root->val; 
        if (root->left == nullptr && root->right == nullptr && curr == targetSum) {
            result = true; 
            return;  
        } 
        if (root->left) {
            traversal(root->left, targetSum); 
        } 
        if (root->right) {
            traversal(root->right, targetSum); 
        }
        curr -= root->val; 
    }
    bool hasPathSum(TreeNode* root, int targetSum) { 
        if (root == nullptr) {
            return false; 
        }
        result = false; 
        traversal(root, targetSum); 
        return result; 
    }
};
```
