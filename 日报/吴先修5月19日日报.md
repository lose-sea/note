## 今日算法题

### 题目1:  [617. 合并二叉树](https://leetcode.cn/problems/merge-two-binary-trees/) 

<img src="/Users/lose_sea/Desktop/pintures/image-20260519213510900.png" alt="image-20260519213510900"  />

```objc

class Solution {
public:
    TreeNode* traversal(TreeNode* root1, TreeNode* root2) {
        if (root1 == nullptr) {
            return root2; 
        } 
        if (root2 == nullptr) {
            return root1; 
        } 
        TreeNode* root = new TreeNode(root1->val + root2->val); 
        root->left = traversal(root1->left, root2->left); 
        root->right = traversal(root1->right, root2->right); 

        return root; 
    }
    TreeNode* mergeTrees(TreeNode* root1, TreeNode* root2) {
        if (root1 == nullptr) {
            return root2; 
        } 
        if (root2 == nullptr) {
            return root1; 
        } 
        return traversal(root1, root2); 
    }
};
```
