## 今日算法题

### 题目1: [404. 左叶子之和](https://leetcode.cn/problems/sum-of-left-leaves/) 

![image-20260512214615226](/Users/lose_sea/Desktop/pintures/image-20260512214615226.png)

```objc
class Solution {
public:
    int sumOfLeftLeaves(TreeNode* root) {
        if (root == nullptr) { 
            return 0; 
        } 
        int sum = 0; 
        stack<TreeNode*> stk; 
        //  前序遍历 
        stk.push(root); 
        while (!stk.empty()) {
            TreeNode* node = stk.top(); 
            if (node != nullptr) {
                stk.pop(); 
                if (node->right) {
                    stk.push(node->right);
                } 
                stk.push(node); 
                stk.push(nullptr); 
                if (node->left) {
                    stk.push(node->left);  
                    TreeNode* temp = node->left; 
                    if (temp->left == nullptr && temp->right == nullptr) {
                        sum += temp->val; 
                    }
                } 
            } else {
                stk.pop(); 
                stk.pop(); 
            } 
        }  
        return sum; 
    }
};
```
