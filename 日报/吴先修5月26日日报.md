```objc
// 采用递归
class Solution {
public:
    int depth = 0; 
    int result = 0; 
    void path(TreeNode* root, int curr) {
        if (root == nullptr) {
            return; 
        }   
        if (curr > depth) { 
            depth = curr; 
            result = root->val; 
        } 
        path(root->left, curr + 1); 
        path(root->right, curr + 1); 
    }
    int findBottomLeftValue(TreeNode* root) {  
        path(root, 1); 
        return result; 
   }
};
```
