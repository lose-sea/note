### 题目1: [138. 随机链表的复制](https://leetcode.cn/problems/copy-list-with-random-pointer/) 

![image-20260429172503696](/Users/lose_sea/Desktop/pintures/image-20260429172503696.png)

```objc
#define ll long long 
class Solution {
public:
    Node* copyRandomList(Node* head) {
        if (head == nullptr) {
            return nullptr; 
        } 
        unordered_map<Node*, Node*> mp; 
        Node* cur = head;  
        while (cur != nullptr) {
            mp[cur] = new Node(cur->val);  
            cur = cur->next; 
        } 
        cur = head; 
        while (cur != nullptr) {
            mp[cur]->next = mp[cur->next];  
            mp[cur]->random = mp[cur->random];  
            cur = cur->next; 
        } 
        return mp[head]; 
    }
}; 
```
