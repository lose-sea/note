## 今日算法题

### 题目1: [128. 最长连续序列](https://leetcode.cn/problems/longest-consecutive-sequence/)

![image-20260323212612096](/Users/lose_sea/Library/Application Support/typora-user-images/image-20260323212612096.png)

```objc
// 排序 O(ologn)
class Solution {
public:
    int longestConsecutive(vector<int>& nums) {
        if (nums.size() == 0) {
            return 0; 
        }  
        sort(nums.begin(), nums.end()); 
        int result = 1; 
        int curr = 1; 
        for (int i = 1; i < nums.size(); i++) {
            if (nums[i] == nums[i - 1]) {
                continue; 
            } else if (nums[i] == nums[i - 1] + 1) {
                curr++; 
            } else {
                curr = 1; 
            } 
            result = max(result, curr); 
        } 
        return result;                                                       
    }
}; 

// 哈希表 O(n)
class Solution {
public:
    int longestConsecutive(vector<int>& nums) {
        if (nums.size() == 0) {
            return 0; 
        }              
        unordered_set<int> st(nums.begin(), nums.end());   
        int result = 0; 
        for (int x : st) {
            if (st.contains(x - 1)) {
                continue; 
            } 
            int y = x + 1; 
            while (st.contains(y)) {
                y++; 
            } 
            result = max(result, y - x); 
        }                                 
        return result;  
    }
}; 
```
