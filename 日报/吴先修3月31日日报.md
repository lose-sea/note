## 题目1: [347. 前 K 个高频元素](https://leetcode.cn/problems/top-k-frequent-elements/) 

![image-20260331205523146](/Users/lose_sea/Library/Application Support/typora-user-images/image-20260331205523146.png)

```objc
class Solution {
public:
    vector<int> topKFrequent(vector<int>& nums, int k) {
        vector<int> result; 
        unordered_map<int, int> mp;  
        int num = 0; 
        for (int i : nums) {
            mp[i]++;  
            num = max(num, mp[i]); 
        }
        vector<vector<int>> buckets(num + 1);  
        for (auto& [x, c] : mp) {
            buckets[c].push_back(x); 
        }         
        for (int i = num; result.size() < k; i--) {
            result.insert(result.end(), buckets[i].begin(), buckets[i].end()); 
        } 
        return result; 
    }
};
```
