## 今日算法题

### 题目1: [98. 验证二叉搜索树](https://leetcode.cn/problems/validate-binary-search-tree/) 

![image-20260602215002556](/Users/lose_sea/Desktop/pintures/image-20260602215002556.png)

```objc
class Solution {
public:

    bool judge = true; 
    TreeNode* pre = nullptr; 
    void traversal(TreeNode* root) {
        if (judge == false) {
            return; 
        } 
        if (root == nullptr) {
            return; 
        }
        traversal(root->left); 
        if (pre != nullptr && root->val <= pre->val) {
            judge = false; 
            return; 
        } 
        pre = root; 
        traversal(root->right); 
    }
    bool isValidBST(TreeNode* root) {
        if (root == nullptr) {
            return true; 
        } 
        traversal(root);
        return judge; 
    }
};
```
