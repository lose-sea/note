## 今日算法题

### 题目1:  [530. 二叉搜索树的最小绝对差](https://leetcode.cn/problems/minimum-absolute-difference-in-bst/)

![image-20260603214240571](/Users/lose_sea/Desktop/pintures/image-20260603214240571.png)

```objc
class Solution {
public: 
    int result = INT_MAX; 
    TreeNode* pre = nullptr; 
    void traversal(TreeNode* root) {
        if (root == nullptr) {
            return; 
        } 
        traversal(root->left); 
        if (pre != nullptr)  {
            int curr = abs(root->val - pre->val); 
            if (curr < result) {
            result = curr; 
        } 
        }
        pre = root; 
        traversal(root->right); 
    }

    int getMinimumDifference(TreeNode* root) {
        traversal(root); 
        return result; 
    }
};
```
