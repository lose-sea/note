## 今日算法题

### 题目1:  [23. 合并 K 个升序链表](https://leetcode.cn/problems/merge-k-sorted-lists/)  

![image-20260421211235038](/Users/lose_sea/Desktop/pintures/image-20260421211235038.png)

```objc
class Solution {
public: 
    ListNode* mergeList(ListNode* l1, ListNode* l2) {
        if (l1 == nullptr) {
            return l2; 
        } 
        if (l2 == nullptr) {
            return l1; 
        } 
        ListNode* dummy = new ListNode(0);  
        ListNode* curr = dummy; 
        while(l1 && l2) {
            if (l1->val <= l2->val) {
                curr->next = l1;  
                l1 = l1->next; 
            } else {
                curr->next = l2; 
                l2 = l2->next; 
            } 
            curr = curr->next; 
        } 
        if (l1 == nullptr) {
            curr->next = l2; 
        } else {
            curr->next = l1; 
        } 
        return dummy->next; 
    }
    ListNode* mergeKLists(vector<ListNode*>& lists) {
        if (lists.empty()) {
            return nullptr; 
        }  
        ListNode* head = nullptr; 
        for (int i = 0; i < lists.size(); i++) {
            head = mergeList(head, lists[i]); 
        }
        return head; 
    }
};
```
