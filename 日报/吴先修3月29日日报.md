## 题目1:

![image-20260329200152210](/Users/lose_sea/Library/Application Support/typora-user-images/image-20260329200152210.png)

```objc
class Solution {
public:
    int lengthOfLongestSubstring(string s) {
        unordered_set<char> st;  
        int left = 0; 
        int curr = 0; 
        int result = 0;
        for (int i = 0; i < s.size(); i++) {
            if (st.find(s[i]) != st.end()) { 
                result = max(curr, result); 
                while (st.find(s[i]) != st.end()) {
                    st.erase(s[left]);  
                    left++; 
                    curr--;     
                }   
            } 
            st.insert(s[i]);  
            curr++; 
        } 
        result = max(result, curr);  
        return result; 
    }
};
```
