```objc
class Solution {
public:
    vector<int> productExceptSelf(vector<int>& nums) {
        // multiply v.乘积 
        vector<int> product_right(nums.size(), 1); 
        for (int i = nums.size() - 2; i >= 0; i--) {
            product_right[i] = product_right[i + 1] * nums[i + 1]; 
        }         
        int l = 1; 
        vector<int> result; 
        for (int i = 0; i < nums.size(); i++) {
            result.push_back(l * product_right[i]); 
            l *= nums[i]; 
        }
        return result; 
    }
}; 

```
