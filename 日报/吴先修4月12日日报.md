## 今日算法题

### 题目1:   [438. 找到字符串中所有字母异位词](https://leetcode.cn/problems/find-all-anagrams-in-a-string/) 

![image-20260412213801495](/Users/lose_sea/Library/Application Support/typora-user-images/image-20260412213801495.png)

```objc
class Solution {
public:
    vector<int> findAnagrams(string s, string p) { 
        vector<int> result; 
        if (s.size() < p.size()) {
            return result; 
        } 
        vector<int> pcount(26); 
        vector<int> scount(26);  
        int len = p.size(); 
        for (int i = 0; i < len; i++) {
            scount[s[i] - 'a']++; 
            pcount[p[i] - 'a']++; 
        }  
        if (pcount == scount) {
            result.push_back(0); 
        } 
        for (int i = 0; i < s.size() - len; i++) {
            scount[s[i] - 'a']--; 
            scount[s[i + len] - 'a']++;  
            if (scount == pcount) {
                result.push_back(i + 1); 
            }
        } 
        return result; 
    }
};
```
