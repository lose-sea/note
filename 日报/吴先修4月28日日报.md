```objc
// 归并排序
#define ll long long 
class Solution {
public:
    ListNode* sortList(ListNode* head) {
        return sortList(head, nullptr); 
    } 
    ListNode* sortList(ListNode* head, ListNode* tail) {
        if (head == nullptr) {
            return head; 
        } 
        if (head->next == tail) {
            head->next = nullptr; 
            return head; 
        }
        ListNode* slow = head; 
        ListNode* fast = head; 
        while (fast != tail) {
            slow = slow->next; 
            fast = fast->next;   
            if (fast != tail) {
                fast = fast->next;  
            }
        } 
        ListNode* mid = slow; 
        return merge(sortList(head, mid), sortList(mid, tail)); 
    } 

    ListNode* merge(ListNode* l1, ListNode* l2) {
        if (l1 == nullptr) {
            return l2; 
        } 
        if (l2 == nullptr) {
            return l1; 
        } 
        ListNode* dummy = new ListNode(0); 
        ListNode* curr = dummy; 
        while (l1 && l2) {
            if (l1->val <= l2->val) {
                curr->next = l1; 
                l1 = l1->next; 
            } else {
                curr->next = l2; 
                l2 = l2->next; 
            } 
            curr = curr->next; 
        }  
        if (l1 == nullptr) {
            curr->next = l2; 
        } else {
            curr->next = l1; 
        }
        return dummy->next; 
    }
};
```
