## 二维数组的"方寸之间"

数组与指针, 你能分清楚吗

```c
#include <stdio.h>
int main() {
    int a[2][3] = {{1,2,3},
                   {4,5,6}};
    int *p = a[0];
    int (*q)[3] = a;
    
    printf("%d\n", *(p + 3) + *(*(q + 1) + 1));
    printf("%d\n", *(a[1] - 1) + *(a[0] + 4));
    printf("%zu\n", sizeof(a + 1) + sizeof(*a));
    printf("%d\n", (p + 2)[1] - (*(q + 1))[-1]);
    return 0;
}
```





## 字符数组的"三重门"

小心字符数组里的隐藏陷阱

```c
#include <stdio.h>
#include <string.h>

void transform(char *s) {
    char *p = s + strlen(s) - 1;
    while (p > s) {
        *p = *(p - 1);
        p--;
    }
    *s = 'X';
}

int main() {
    char str[20] = "3G";
    char *ptr = str + 1;
    
    strcpy(str + 2, "lab");
    transform(str);
    strcat(ptr, "2025");
    
    printf("%d\n", strlen(str) - sizeof(str) / sizeof(char));
    printf("%c\n", *(str + 3));
    printf("%s\n", ptr + 1);
    
    return 0;
}
```
